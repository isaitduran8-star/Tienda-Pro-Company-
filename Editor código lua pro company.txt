-- Crear carpetas
function crearCarpetas()
    local folder = File(carpetaBase)
    if not folder.exists() then
        folder.mkdirs()
    end
    local folderArchivos = File(carpetaArchivos)
    if not folderArchivos.exists() then
        folderArchivos.mkdirs()
    end
end

-- Cargar preferencia de no mostrar bienvenida
local prefs = context.getSharedPreferences("EditorLuaPro", Context.MODE_PRIVATE)
noMostrarBienvenida = prefs.getBoolean("noMostrarBienvenida", false)
vibracionActivada = prefs.getBoolean("vibracionActivada", true)
intensidadVibracion = prefs.getInt("intensidadVibracion", 500)
betaAceptado = prefs.getBoolean("betaAceptado", false)

-- Función para guardar preferencias
function guardarPreferencia(noMostrar)
    local editor = prefs.edit()
    editor.putBoolean("noMostrarBienvenida", noMostrar)
    editor.commit()
end

function guardarPreferenciasVibracion(activada, intensidad)
    local editor = prefs.edit()
    editor.putBoolean("vibracionActivada", activada)
    editor.putInt("intensidadVibracion", intensidad)
    editor.commit()
    vibracionActivada = activada
    intensidadVibracion = intensidad
end

function guardarBetaAceptado(aceptado)
    local editor = prefs.edit()
    editor.putBoolean("betaAceptado", aceptado)
    editor.commit()
    betaAceptado = aceptado
end

-- Función para vibrar
function vibrar(ms)
    if vibracionActivada then
        local tiempo = ms or intensidadVibracion
        vibrator.vibrate(tiempo)
    end
end

-- Función para mostrar Toast
function showToast(mensaje, duracion)
    duracion = duracion or Toast.LENGTH_SHORT
    Toast.makeText(context, mensaje, duracion).show()
end

function mostrarGestorArchivos()
    local archivos = {}
    local folder = File(carpetaArchivos)
    
    if folder.exists() and folder.isDirectory() then
        local listaJava = folder.listFiles()
        if listaJava then
            local longitud = #listaJava
            for i = 0, longitud - 1 do
                local archivo = listaJava[i]
                if archivo.isFile() then
                    local nombre = archivo.getName()
                    if string.match(nombre, "%.txt$") then
                        table.insert(archivos, nombre)
                    end
                end
            end
        end
    end
    
    if #archivos == 0 then
        vibrar(200)
        showToast("📂 No hay archivos guardados")
        return
    end
    
    local layout = {
        LinearLayout;
        orientation = "vertical";
        padding = "10dp";
        {
            TextView;
            text = "📂 ARCHIVOS GUARDADOS";
            textSize = "16sp";
            gravity = "center";
            textColor = "#2196F3";
            padding = "10dp";
        };
        {
            TextView;
            text = "Carpeta: " .. carpetaArchivos;
            textSize = "11sp";
            gravity = "center";
            textColor = "#666666";
            padding = "5dp";
        };
        {
            ListView;
            id = "listaArchivos";
            layout_width = "match_parent";
            layout_height = "300dp";
        };
        {
            Button;
            text = "❌ CERRAR";
            layout_width = "match_parent";
            backgroundColor = "#F44336";
            textColor = "#FFFFFF";
            padding = "15dp";
            onClick = function()
                vibrar(300)
                if fileManagerDialog then
                    fileManagerDialog.dismiss()
                    fileManagerDialog = nil
                end
            end
        };
    }
    
    fileManagerDialog = LuaDialog(context)
    fileManagerDialog.setTitle("📂 Gestor de Archivos")
    fileManagerDialog.setView(loadlayout(layout))
    fileManagerDialog.setCancelable(true)
    fileManagerDialog.show()
    
    local adapter = ArrayAdapter(context, android.R.layout.simple_list_item_1, archivos)
    listaArchivos.setAdapter(adapter)
    
    listaArchivos.setOnItemClickListener{
        onItemClick = function(parent, view, position, id)
            vibrar(300)
            local nombreArchivo = archivos[position + 1]
            if fileManagerDialog then
                fileManagerDialog.dismiss()
                fileManagerDialog = nil
            end
            mostrarOpcionesArchivo(nombreArchivo)
        end
    }
end

function cargarArchivo(nombreArchivo)
    local ruta = carpetaArchivos .. "/" .. nombreArchivo
    local file = io.open(ruta, "r")
    if file then
        local contenido = file:read("*all")
        file:close()
        textoCompleto.setText(contenido)
        texto_completo = contenido
        procesar_texto()
        vibrar(300)
        showToast("✅ Archivo cargado: " .. nombreArchivo)
    else
        vibrar(200)
        showToast("❌ Error al cargar")
    end
end

function eliminarArchivo(nombreArchivo)
    local ruta = carpetaArchivos .. "/" .. nombreArchivo
    local file = File(ruta)
    if file.exists() then
        if file.delete() then
            vibrar(300)
            showToast("🗑 Archivo eliminado: " .. nombreArchivo)
            if fileManagerDialog then
                fileManagerDialog.dismiss()
                fileManagerDialog = nil
            end
            mostrarGestorArchivos()
        else
            vibrar(200)
            showToast("❌ Error al eliminar")
        end
    end
end

function procesar_texto()
    texto_completo = tostring(textoCompleto.getText())
    
    if #texto_completo == 0 then
        vibrar(200)
        showToast("❌ No hay código para procesar")
        return
    end
    
    bloques = {}
    local total = #texto_completo
    total_bloques = math.ceil(total / tamano_bloque)
    
    for i = 1, total_bloques do
        local inicio = (i - 1) * tamano_bloque + 1
        local fin = math.min(i * tamano_bloque, total)
        bloques[i] = string.sub(texto_completo, inicio, fin)
    end
    
    mostrar_bloques()
    
    vibrar(300)
    showToast(string.format("✅ %d bloques de código creados", total_bloques))
end

function mostrar_bloques()
    tablaBloques.removeAllViews()
    tituloBloques.setVisibility(0)
    
    local row = nil
    local contador = 0
    
    for i = 1, #bloques do
        if contador % 3 == 0 then
            row = TableRow(context)
            row.setLayoutParams(TableRow.LayoutParams(TableRow.LayoutParams.MATCH_PARENT, TableRow.LayoutParams.WRAP_CONTENT))
            tablaBloques.addView(row)
        end
        
        local btn = Button(context)
        local texto_bloque = bloques[i]
        local preview = string.sub(texto_bloque, 1, 20)
        if #texto_bloque > 20 then
            preview = preview .. "..."
        end
        
        btn.setText(string.format("📦 Bloque %d\n(%d chars)\n%s", i, #texto_bloque, preview))
        btn.setTextSize(12)
        btn.setPadding(5, 5, 5, 5)
        btn.setAllCaps(false)
        
        local params = TableRow.LayoutParams(0, TableRow.LayoutParams.WRAP_CONTENT, 1.0)
        params.setMargins(5, 5, 5, 5)
        btn.setLayoutParams(params)
        
        btn.onClick = function()
            vibrar(300)
            editar_bloque(i)
        end
        
        row.addView(btn)
        contador = contador + 1
    end
end

            textSize = "14sp";
            padding = "5dp";
        };
        {
            ScrollView;
            layout_width = "match_parent";
            layout_height = "250dp";
            {
                EditText;
                id = "outputText";
                hint = "Resultado de la ejecución...";
                textSize = "14sp";
                gravity = "top";
                layout_width = "match_parent";
                layout_height = "match_parent";
                maxLines = 100;
                focusable = false;
                textColor = "#4CAF50";
            };
        };
        {
            LinearLayout;
            orientation = "horizontal";
            layout_width = "match_parent";
            {
                Button;
                text = "📋 COPIAR";
                layout_width = "0dp";
                layout_weight = "1";
                onClick = function()
                    local texto = outputText.getText():toString()
                    if #texto > 0 then
                        local clipboard = context.getSystemService(Context.CLIPBOARD_SERVICE)
                        local clip = android.content.ClipData.newPlainText("Resultado", texto)
                        clipboard.setPrimaryClip(clip)
                        vibrar(300)
                        showToast("✅ Resultado copiado")
                    end
                end
            };
            {
                Button;
                text = "❌ CERRAR";
                layout_width = "0dp";
                layout_weight = "1";
                onClick = function()
                    vibrar(300)
                    if execDialog then
                        execDialog.dismiss()
                        execDialog = nil
                    end
                end
            };
        };
    }
    
    execDialog = LuaDialog(context)
    execDialog.setTitle("▶️ Ejecutando Código Lua")
    execDialog.setView(loadlayout(layout))
    execDialog.setCancelable(true)
    execDialog.show()
    
    local output = ""
    local success, err = pcall(function()
        local oldPrint = print
        local outputLines = {}
        
        print = function(...)
            local args = {...}
            local line = table.concat(args, " ")
            table.insert(outputLines, line)
            output = table.concat(outputLines, "\n")
            if outputText then
                outputText.setText(output)
            end
        end
        
        local func, loadErr = loadstring(texto_completo)
        if func then
            local result = func()
            if result ~= nil then
                if type(result) == "table" then
                    local tableStr = {}
                    for k, v in pairs(result) do
                        table.insert(tableStr, string.format("%s = %s", tostring(k), tostring(v)))
                    end
                    print("Return: " .. table.concat(tableStr, ", "))
                else
                    print("Return: " .. tostring(result))
                end
            end
        else
            print("Error de compilación: " .. tostring(loadErr))
        end
        
        print = oldPrint
    end)
    
    if not success then
        output = "❌ ERROR: " .. tostring(err) .. "\n\n" .. output
    end
    
    if #output == 0 then
        output = "✅ Código ejecutado correctamente (sin salida)"
    end
    
    if outputText then
        outputText.setText(output)
    end
    
    vibrar(500)
    showToast("✅ Ejecución completada")
end
